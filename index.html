// CODICE PER CLOUDFLARE WORKERS (Versione con Classifiche Reali) const API_KEY = "e0ca06c07c634d4fb0950365bd82ffd0"; // La tua API Key di Football-Data const CAMPIONATI = ["WC", "CL", "BL1", "DED", "FL1", "PD", "SA", "PL", "ELC", "PPL", "BSA", "CLI"]; const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "GET, HEAD, POST, OPTIONS", "Access-Control-Allow-Headers": "Content-Type", "Content-Type": "application/json" }; export default { async fetch(request, env, ctx) { if (request.method === "OPTIONS") { return new Response(null, { headers: corsHeaders }); } try { const dateFrom = new Date().toISOString().split('T')[0]; const dateTo = new Date(Date.now() + 4 * 24 * 60 * 60 * 1000).toISOString().split('T')[0]; let tutteLePartite = []; for (const cod of CAMPIONATI) { try { // 1. Scarichiamo la classifica del campionato const resStandings = await fetch(`https://api.football-data.org/v4/competitions/${cod}/standings`, { headers: { "X-Auth-Token": API_KEY } }); let tabellaMappa = {}; if (resStandings.ok) { const dataStandings = await resStandings.json(); // Creiamo una mappa rapida per cercare le squadre in base al loro ID if (dataStandings.standings && dataStandings.standings[0]) { const table = dataStandings.standings[0].table || []; table.forEach(row => { tabellaMappa[row.team.id] = { posizione: row.position, punti: row.points, giocate: row.playedGames, golFatti: row.goalsFor, golSubiti: row.goalsAgainst }; }); } } // 2. Scarichiamo le partite dei prossimi 4 giorni per questo campionato const resMatches = await fetch(`https://api.football-data.org/v4/competitions/${cod}/matches?dateFrom=${dateFrom}&dateTo=${dateTo}`, { headers: { "X-Auth-Token": API_KEY } }); if (resMatches.ok) { const dataMatches = await resMatches.json(); const matches = (dataMatches.matches || []).filter(m => { const st = (m.status || "").toUpperCase(); return st === "TIMED" || st === "SCHEDULED"; }); // 3. Uniamo i dati della partita con le statistiche reali di classifica matches.forEach(m => { const statsCasa = tabellaMappa[m.homeTeam.id] || { punti: "N/D", posizione: "N/D", giocate: "N/D", golFatti: "N/D", golSubiti: "N/D" }; const statsOspite = tabellaMappa[m.awayTeam.id] || { punti: "N/D", posizione: "N/D", giocate: "N/D", golFatti: "N/D", golSubiti: "N/D" }; tutteLePartite.push({ campionato: m.competition ? m.competition.name : "N/D", squadraCasa: m.homeTeam.name, squadraOspite: m.awayTeam.name, dataOraUTC: m.utcDate, puntiCasa: statsCasa.punti, puntiOspite: statsOspite.punti, posizioneCasa: statsCasa.posizione, posizioneOspite: statsOspite.posizione, giocateCasa: statsCasa.giocate, giocateOspite: statsOspite.giocate, golFattiCasa: statsCasa.golFatti, golFattiOspite: statsOspite.golFatti, golSubitiCasa: statsCasa.golSubiti, golSubitiOspite: statsOspite.golSubiti }); }); } } catch (e) { console.log(`Errore nel campionato ${cod}:`, e); } } return new Response(JSON.stringify({ errore: false, totalePartiteTrovate: tutteLePartite.length, partite: tutteLePartite }), { headers: corsHeaders }); } catch (err) { return new Response(JSON.stringify({ errore: true, messaggio: err.message }), { status: 500, headers: corsHeaders }); } } };
